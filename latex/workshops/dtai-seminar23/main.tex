		\textbf{What is a \emph{good} explanation ?} \pause \\
		
		Let $\mathbf{f}  ($ \includegraphics[width=10pt]{figures/explanation_steps/explanation_E}, {\includegraphics[width=10pt]{figures/explanation_steps/explanation_S}}, {\includegraphics[width=10pt]{figures/explanation_steps/explanation_N}} $)$ be a cost-function that quantifies how good (e.g. easy to understand) an explanation step is.
		\pause 
		
		\vspace*{1cm}
		
		\textbf{What is the \emph{best/easiest} explanation step? }\pause
		\begin{algorithm}[H]
			
			$\mathtt{expl}_{\text{best}} \gets \mathit{nil}$\;\pause
			\For{remaining assignment {\includegraphics[width=10pt]{figures/explanation_steps/explanation_N}} to explain}{
				
				expl $\gets \mathtt{\mathbf{MUS}}($ \includegraphics[width=10pt]{figures/explanation_steps/explanation_E_full} $\wedge$ {\includegraphics[width=10pt]{figures/explanation_steps/explanation_S_full}} $\wedge \lnot$ {\includegraphics[width=10pt]{figures/explanation_steps/explanation_N}} $, f)$\; \pause
				\uIf{$f(\mathtt{expl})<f(\mathtt{expl}_{\text{best}})$}{
					$\mathtt{expl}_{\text{best}} \gets \mathtt{expl}$\;
				}
			}
			\Return{$\mathtt{expl}_{\text{best}}$} 
		\end{algorithm}
\end{frame}
