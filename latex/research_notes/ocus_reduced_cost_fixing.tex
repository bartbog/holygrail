	\section{Optimal Constrained Unsatisfiable Subsets}\label{sec:OCUS}
	
	\begin{algorithm}[!h]
		\DontPrintSemicolon
		$\setstohit  \gets \emptyset$ \; %\label{omus-line1} 
		$LB \gets 0; UB \gets \infty$\;
		\While{$LB < UB$}{
			$hs\_is\_opt, \m{S} \gets \cohs(\setstohit,f,p) $ \label{line:opt} \;
			\lIf{hs\_is\_opt}{$LB \gets f(\m{S})$}
			\lIf{ $\lnot \sat(\m{S})$ \label{alg:ocus-sat-check}}{\Return{$\m{S}$}}

			$\m{K}, \m{S}' \gets \corrsubsets(\subsetT, \formula)$ \label{line:grow}\;
			$\setstohit  \gets \setstohit  \cup \m{K} $  \label{alg:ocus:complement}\;
		}
		\caption{$\comus(\formula,f,p)$ }
		\label{alg:comus}
	\end{algorithm}
	
	
	\begin{algorithm}[!h]
		\DontPrintSemicolon
		$\m{K} \gets \emptyset$\;
		$\subsetT' \gets  \subsetT$ \label{alg:disjmcs:init} \;
		
		\While{$\sat(\m{S}')$}{
			$\m{C} \gets  \formula \setminus \grow(\m{S}', \formula)$ \label{alg:corr-subsets-grow-with-sat:corr-subset}\;
			$ \m{S}' \gets \m{S}'  \cup  \m{C} $\label{alg:corr-subsets-grow-with-sat:mapping}\;
			$ \m{K} \gets \m{K} \cup \{\m{C} \}$\;
		}
		\Return $\m{K}, m{S}'$
		\caption{$\corrsubsets(\subsetT, \formula)$ }
		\label{alg:corr-subsets-grow-with-sat-old}
	\end{algorithm}

	\section{Multiple Optimal Constrained Unsatisfiable Subsets}\label{sec:OCUS}
	
	\begin{algorithm}[!ht]
		\DontPrintSemicolon
		\caption{$\onestepomusbounded(\formulac,f,I,\Iend)$}
		\label{alg:oneStepBoundBart}
		$\m{S}_{best} \gets \mathit{nil}$ \;
		${\cal S}^l_{best} \gets \mathit{nil}$ for each $l$ \quad \quad (or from previous iteration) \;
		\For{$l \in \{\Iend \setminus I\}$ sorted by $f({\cal S}^l_{best})$\label{alg:oneStepBoundBart:sorting}}{
			${\cal S}^l_{best}, \mathit{status} \gets \call{OCUS}{( (\formulac \land I \land \neg l), f, f(\m{S}) \leq f(\m{S}_{best}) )}$\label{alg:oneStepBound:cost_check}\;
			\If{$\mathit{status} \neq \mathit{FAILURE}$ }{
				$\m{S}_{best} \gets \m{S}^l_{best}$\;
			}
		}
		\Return{${\cal S}_{best}$} 
	\end{algorithm}
	
	
	\begin{algorithm}[!ht]
		\DontPrintSemicolon
		\caption{$\iterativeonestep(\formulac,f,I,\Iend)$}
		\label{alg:oneStepIter}
		$queue \gets$ \texttt{InitializePriorityQueue($(\ell, \emptyset,\emptyset): 0 \ | \ \forall \ell \in \Iend \setminus I$)}\; %// init with $(l, \emptyset): 0$ for each $l$, or from previous iteration \;
		%$S_{best} \gets \mathit{nil}$; $v_{best} \gets \infty$\;
		%${\cal S}^l_{best} \gets \mathit{nil}$ for each $l$ // or from previous iteration \;
		\While{$(\ell, \m{S}_\ell, \setstohit_{\ell}) \gets queue.pop()$}{
			%\For{$l \in \{\Iend \setminus I\}$ sorted by $f({\cal S}^l_{best})$}{
				\If{ $\lnot \sat(\m{S}_\ell)$}{\label{alg:onestepIter-sat-check}
					\Return{$\m{S}_\ell$} \;
				}
				
				$\m{K} \gets \corrsubsets(\subsetT_\ell, (\formulac \land I \land \neg \ell))$\;
				$\setstohit_\ell  \gets \setstohit_\ell  \cup \m{K} $\;
				$\m{S}_\ell \gets OptHittingSet(\setstohit_\ell,f) $  \;
				%			$\m{S} \gets \call{OUS\_oneIter}{( (\formulac \land I \land \neg l), f,\m{S})}$\;
				\texttt{queue.push($(\ell, \m{S}_\ell, \setstohit_{\ell}): f(\m{S_\ell})$)}\;
			}
		\end{algorithm}
